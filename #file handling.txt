# Writing to a file
file = open('example.txt', 'w')

# Appending to the file (use 'a' mode)
with open('example.txt', 'a') as file:
    file.write('to work with keyword\n')

# Reading the file
with open('example.txt', 'r') as file:
    content = file.read()
    print("File content:")
    print(content)

finally:
